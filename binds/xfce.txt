# Or run: karui-oto binds --apply xfce

# Media keys: add XF86AudioPlay/Next/Prev -> karui-media {play-pause|next|prev}.

# songs:   key <Super>o -> ~/.local/bin/karui-oto songs
# artists: key <Super>p -> ~/.local/bin/karui-oto artists
# albums:  key <Super><Shift>p -> ~/.local/bin/karui-oto albums
# folders: key <Super><Shift>o -> ~/.local/bin/karui-oto folders
# kill:    key <Super>x -> ~/.local/bin/karui-oto kill
# media-next: key XF86AudioNext -> ~/.local/bin/karui-media next
# media-prev: key XF86AudioPrev -> ~/.local/bin/karui-media prev
# media-play: key XF86AudioPlay -> ~/.local/bin/karui-media play-pause
